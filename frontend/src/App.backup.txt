async function updateAlertStatus(alertId, newStatus) {
  try {
    const response = await fetch(
      `http://127.0.0.1:8000/alerts/${alertId}/status`,
      {
        method: "PATCH",
        headers: {
          "Content-Type": "application/json"
        },
        body: JSON.stringify({ status: newStatus })
      }
    );

    if (!response.ok) {
      throw new Error("Status update failed");
    }

    setAlerts((previousAlerts) =>
      previousAlerts.map((alert) =>
        alert.id === alertId
          ? { ...alert, status: newStatus }
          : alert
      )
    );

  } catch (error) {
    window.alert("Could not update alert status");
  }
}

async function analyzeThreat() {
  setMlLoading(true);
  setMlResult(null);

  try {
    const response = await fetch(
      "http://127.0.0.1:8000/predict-and-save",
      {
        method: "POST",
        headers: {
          "Content-Type": "application/json"
        },
        body: JSON.stringify({
          failed_logins: Number(mlForm.failed_logins),
          successful_logins: Number(mlForm.successful_logins),
          unique_users: Number(mlForm.unique_users),
          username: mlForm.username,
          source_ip: mlForm.source_ip
        })
      }
    );

    if (!response.ok) {
      throw new Error("Prediction failed");
    }

    const result = await response.json();
    setMlResult(result);

    if (result.saved) {
      const alertsResponse = await fetch(
        "http://127.0.0.1:8000/alerts"
      );

      if (alertsResponse.ok) {
        const updatedAlerts = await alertsResponse.json();
        setAlerts(updatedAlerts);
      }
    }

  } catch (error) {
    setMlResult({
      error: "Could not analyze threat. Check FastAPI."
    });
  } finally {
    setMlLoading(false);
  }
}

function downloadAlertsCSV() {
  const headers = [
    "ID",
    "Alert Type",
    "Severity",
    "Username",
    "Source IP",
    "Timestamp",
    "Status"
  ];

  const rows = alerts.map((alert) => [
    alert.id,
    alert.alert_type,
    alert.severity,
    alert.username,
    alert.source_ip,
    alert.timestamp,
    alert.status
  ]);

  const escapeCSV = (value) => {
    let text = String(value ?? "");

    if (/^[\s]*[=+\-@]/.test(text)) {
      text = "'" + text;
    }

    return `"${text.replace(/"/g, '""')}"`;
  };

  const csv = [headers, ...rows]
    .map((row) => row.map(escapeCSV).join(","))
    .join("\r\n");

  const blob = new Blob([csv], {
    type: "text/csv;charset=utf-8;"
  });

  const url = URL.createObjectURL(blob);
  const link = document.createElement("a");

  link.href = url;
  link.download = "SOC_Security_Alerts.csv";
  link.click();

  URL.revokeObjectURL(url);

  ML Prediction: {mlResult.prediction}
</p>

          
<p>
  {mlResult.duplicate
    ? `Duplicate Alert Prevented! Existing Alert ID: ${mlResult.alert_id}`
    : mlResult.saved
      ? `Alert saved! ID: ${mlResult.alert_id}`
      : "Normal activity - No alert saved"}
</p>

        </>
      )}
    </div>
  )}
</div>

      
<h2 id="security-alerts" style={{ marginTop: "35px" }}>
        Recent Security Alerts
      </h2>


<button id="reports"
  type="button"
  onClick={downloadAlertsCSV}
  disabled={alerts.length === 0}
  style={{
    background: "#0284c7",
    color: "white",
    padding: "12px 20px",
    border: "none",
    borderRadius: "8px",
    cursor: "pointer",
    marginBottom: "20px"
  }}
>
  Download CSV Report
</button>

      <input
      className="soc-alert-search"
  type="text"
  placeholder="Search by IP, username or attack..."
  value={search}
  onChange={(e) => setSearch(e.target.value)}
  style={{
    padding: "12px",
    width: "320px",
    maxWidth: "100%",
    marginBottom: "20px",
    borderRadius: "8px",
    background: "#1e293b",
    color: "white",
    border: "1px solid #475569"
  }}
/>

<select
  value={severityFilter}
  onChange={(e) => setSeverityFilter(e.target.value)}
  style={{
    padding: "12px",
    marginLeft: "15px",
    borderRadius: "8px",
    background: "#1e293b",
    color: "white",
    border: "1px solid #475569"
  }}
>
  <option value="ALL">All Severities</option>
  <option value="HIGH">High</option>
  <option value="MEDIUM">Medium</option>
  <option value="LOW">Low</option>
</select>

<select

className="soc-filter-select"
  value={statusFilter}
  onChange={(e) => setStatusFilter(e.target.value)}
  style={{
    padding: "12px",
    marginLeft: "12px",
    background: "#1e293b",
    color: "white",
    border: "1px solid #475569",
    borderRadius: "8px"
  }}
>
  <option value="ALL">All Statuses</option>
  <option value="New">New</option>
  <option value="Investigating">Investigating</option>
  <option value="Resolved">Resolved</option>
</select>

<button
className="soc-clear-btn"
  type="button"
  onClick={() => {
    setSearch("");
    setSeverityFilter("ALL");
    setStatusFilter("ALL");
  }}
  style={{
    padding: "12px 20px",
    marginLeft: "12px",
    background: "#dc2626",
    color: "white",
    border: "none",
    borderRadius: "8px",
    cursor: "pointer"
  }}
>
  Clear Filters
</button>

      {loading && <p>Loading alerts...</p>}
      {error && <p style={{ color: "red" }}>{error}</p>}

      {!loading && !error && alerts.length === 0 && (
        <p>No security alerts found.</p>
      )}

      <div style={{ overflowX: "auto" }}>
        <table className="alert-table" style={{
          width: "100%",
          borderCollapse: "collapse"
        }}>
          <thead>
            <tr>
              {["ID", "Alert Type", "Severity",
                "Username", "Source IP", "Status", "Actions"].map(
                (item) => (
                  <th key={item} style={{
                    padding: "14px",
                    background: "#334155",
                    textAlign: "left"
                  }}>
                    {item}
                  </th>
                )
              )}
            </tr>
          </thead>

          
<tbody>
  
{!loading && !error && alerts.length > 0 && filteredAlerts.length === 0 && (
  <tr>
    <td
      colSpan={6}
      style={{
        textAlign: "center",
        padding: "25px",
        color: "#94a3b8"
      }}
    >
      No matching security alerts found.
    </td>
  </tr>
)}

 
{[...filteredAlerts]
  .sort((a, b) => {
    const priority = { HIGH: 3, MEDIUM: 2, LOW: 1 };

    return (
      (priority[b.severity?.toUpperCase()] ?? 0) -
      (priority[a.severity?.toUpperCase()] ?? 0)
    );
  })
  .map((alert) => (

    
<tr
  key={alert.id}
  style={{
    backgroundColor:
      alert.severity?.toUpperCase() === "HIGH"
        ? "rgba(239, 68, 68, 0.25)"
        : "transparent"
  }}
>

      {[
        alert.id,
        alert.alert_type,
        alert.severity,
        alert.username,
        alert.source_ip
      ].map((value, index) => (
        <td
          key={index}
          style={{
            padding: "14px",
            borderBottom: "1px solid #334155"
          }}
        >
          
{index === 2 ? (
  <span className={`badge badge-${String(value).toLowerCase()}`}>
    ● {value}
  </span>
) : (
  value
)}

        </td>
      ))}

      <td style={{ padding: "14px" }}>
        <select
      className={`soc-status-select status-${String(alert.status).toLowerCase()}`}
          value={alert.status}
          onChange={(e) =>
            updateAlertStatus(alert.id, e.target.value)
          }
          style={{
            padding: "10px",
            background: "#1e293b",
            color: "white",
            borderRadius: "8px",
            border: "1px solid #475569"
          }}
        >
          <option value="New">New</option>
          <option value="Investigating">Investigating</option>
          <option value="Resolved">Resolved</option>
        </select>
      </td>
      
<td style={{ padding: "10px" }}>
  <button
    type="button"
    onClick={() => setSelectedAlert(alert)}
    onMouseEnter={(e) => {
  e.currentTarget.style.background = "#263852";
  e.currentTarget.style.boxShadow = "0 0 15px rgba(34, 211, 238, 0.25)";
}}
onMouseLeave={(e) => {
  e.currentTarget.style.background = "#1e293b";
  e.currentTarget.style.boxShadow = "none";
}}
    style={{
      background: "#2563eb",
      color: "white",
      border: "none",
      padding: "8px 12px",
      borderRadius: "6px",
      cursor: "pointer"
    }}
  >
    View Details
  </button>
  className="soc-view-btn"
</td>

    </tr>
  ))}
</tbody>

        </table>
        
{selectedAlert && (
  <div
    style={{
      position: "fixed",
      inset: 0,
      background: "rgba(0, 0, 0, 0.75)",
      display: "flex",
      justifyContent: "center",
      alignItems: "center",
      zIndex: 9999
    }}
    onClick={(e) => {
  if (e.target === e.currentTarget) {
    setSelectedAlert(null);
  }
}}
  >
    <div
      style={{
        background: "#1e293b",
        color: "white",
        padding: "30px",
        borderRadius: "12px",
        width: "90%",
        maxWidth: "450px"
      }}
    >
      <h2>Security Alert Details</h2>

      <p><b>Alert ID:</b> {selectedAlert.id}</p>
      <p><b>Attack Type:</b> {selectedAlert.alert_type}</p>
      <p><b>Severity:</b> {selectedAlert.severity}</p>
      <p><b>Status:</b> {selectedAlert.status}</p>
      <p><b>Username:</b> {selectedAlert.username}</p>
      <p><b>Source IP:</b> {selectedAlert.source_ip}</p>
      <p><b>Timestamp:</b> {selectedAlert.timestamp || "N/A"}</p>

      <button
        onClick={() => setSelectedAlert(null)}
        style={{
          background: "#ef4444",
          color: "white",
          border: "none",
          padding: "10px 20px",
          borderRadius: "6px",
          cursor: "pointer"
        }}
      >
        Close
      </button>
    </div>
  </div>
)}

      </div>
    </main>
  );
}

export default App;
